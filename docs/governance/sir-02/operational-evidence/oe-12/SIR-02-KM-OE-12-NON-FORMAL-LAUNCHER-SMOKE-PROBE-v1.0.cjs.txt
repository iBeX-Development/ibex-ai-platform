const { spawn } = require('node:child_process');
const mode = process.argv[2];
if (mode === 'echo') {
  process.stdout.write('STDOUT_MARKER:' + JSON.stringify(process.argv.slice(3)) + '\n');
  process.stderr.write('STDERR_MARKER\n');
} else if (mode === 'nonzero') {
  process.stdout.write('NONZERO_STDOUT\n');
  process.stderr.write('NONZERO_STDERR\n');
  process.exit(23);
} else if (mode === 'tree') {
  const child = spawn(process.execPath, [__filename, 'hold-child'], { stdio: 'ignore', windowsHide: true });
  child.on('error', (error) => { process.stderr.write('CHILD_ERROR:' + error.message + '\n'); process.exit(24); });
  process.stdout.write('TREE_CHILD_PID=' + child.pid + '\n');
  process.stderr.write('TREE_STARTED\n');
  setInterval(() => {}, 1000);
} else if (mode === 'hold-child') {
  setInterval(() => {}, 1000);
} else {
  process.stderr.write('UNKNOWN_MODE\n');
  process.exit(25);
}
