function Set-Case([string]$id,[string]$expected,[string]$actual,[string]$result) {
  $caseResults[$id] = [ordered]@{id=$id; expected=$expected; actual=$actual; result=$result}
}
function Compare-StringArrays([string[]]$expected,[string[]]$actual) {
  if ($null -eq $expected -or $null -eq $actual -or $expected.Length -ne $actual.Length) { return $false }
  for ($i=0; $i -lt $expected.Length; $i++) {
    if (-not [string]::Equals($expected[$i],$actual[$i],[System.StringComparison]::Ordinal)) { return $false }
  }
  return $true
}
function Quote-WindowsArgument([AllowEmptyString()][string]$value) {
  if ($value.IndexOf([char]0) -ge 0) { throw 'NUL in process argument' }
  $builder = New-Object System.Text.StringBuilder
  $slash = [string][char]92
  $quote = [string][char]34
  [void]$builder.Append($quote)
  $backslashes = 0
  foreach ($ch in $value.ToCharArray()) {
    if ($ch -eq [char]92) { $backslashes++; continue }
    if ($ch -eq [char]34) {
      [void]$builder.Append(($slash * (2 * $backslashes + 1)))
      [void]$builder.Append($quote)
      $backslashes = 0
      continue
    }
    if ($backslashes -gt 0) {
      [void]$builder.Append(($slash * $backslashes))
      $backslashes = 0
    }
    [void]$builder.Append($ch)
  }
  if ($backslashes -gt 0) { [void]$builder.Append(($slash * (2 * $backslashes))) }
  [void]$builder.Append($quote)
  return $builder.ToString()
}

function Invoke-BoundedNode {
  param(
    [Parameter(Mandatory=$true)][string[]]$processArguments,
    [Parameter(Mandatory=$true)][int]$expectedArgumentCount,
    [Parameter(Mandatory=$true)][int]$timeoutMs,
    [Parameter(Mandatory=$true)][string]$stageName
  )
  $preflight = [ordered]@{
    stage=$stageName
    utc=([DateTime]::UtcNow).ToString('o')
    argumentVariableName=$argumentVariableName
    automaticArgsReferenceCount=$automaticArgsReferenceCount
    executablePath=$node
    executableExists=(Test-Path -LiteralPath $node -PathType Leaf)
    probePath=$probe
    probeExists=(Test-Path -LiteralPath $probe -PathType Leaf)
    expectedArgumentCount=$expectedArgumentCount
    actualPrelaunchArgumentCount=$processArguments.Length
    arguments=@($processArguments)
    probeFirst=[string]::Equals($processArguments[0],$probe,[System.StringComparison]::Ordinal)
    argumentsString=$null
  }
  if (-not $preflight.executableExists -or [string]::IsNullOrWhiteSpace($node)) { throw "Pre-launch executable invalid: $stageName" }
  if (-not $preflight.probeExists -or [string]::IsNullOrWhiteSpace($probe)) { throw "Pre-launch probe missing: $stageName" }
  if ($automaticArgsReferenceCount -ne 0 -or $argumentVariableName -eq 'args') { throw "Automatic argument variable used: $stageName" }
  if ($processArguments.Length -eq 0 -or $processArguments.Length -ne $expectedArgumentCount -or -not $preflight.probeFirst) { throw "Pre-launch argument count/probe mismatch: $stageName" }
  $quoted = New-Object System.Collections.Generic.List[string]
  for ($i=0; $i -lt $processArguments.Length; $i++) {
    $quoted.Add((Quote-WindowsArgument -value $processArguments[$i]))
  }
  $constructedArguments = [string]::Join(' ', $quoted.ToArray())
  $preflight.argumentsString = $constructedArguments
  [void]$preflightHistory.Add($preflight)
  if ([string]::IsNullOrWhiteSpace($constructedArguments)) { throw "Pre-launch Arguments string empty: $stageName" }
  if (-not $constructedArguments.StartsWith((Quote-WindowsArgument -value $probe),[System.StringComparison]::Ordinal)) { throw "Pre-launch probe path absent from invocation: $stageName" }
  $psi = New-Object System.Diagnostics.ProcessStartInfo
  $psi.FileName = $node
  $psi.Arguments = $constructedArguments
  $psi.WorkingDirectory = $repo
  $psi.UseShellExecute = $false
  $psi.CreateNoWindow = $true
  $psi.RedirectStandardOutput = $true
  $psi.RedirectStandardError = $true
  $psi.StandardOutputEncoding = [System.Text.UTF8Encoding]::new($false)
  $psi.StandardErrorEncoding = [System.Text.UTF8Encoding]::new($false)
  $process = New-Object System.Diagnostics.Process
  $process.StartInfo = $psi
  $stages = New-Object System.Collections.ArrayList
  $stageStart = [DateTime]::UtcNow
  [void]$stages.Add([ordered]@{name='Prepared'; utc=$stageStart.ToString('o')})
  if (-not $process.Start()) { throw "Process.Start returned false: $stageName" }
  [void]$stages.Add([ordered]@{name='Started'; utc=([DateTime]::UtcNow).ToString('o')})
  $pidValue = $process.Id
  $outTask = $process.StandardOutput.ReadToEndAsync()
  $errTask = $process.StandardError.ReadToEndAsync()
  $exited = $process.WaitForExit($timeoutMs)
  $timedOut = -not $exited
  $killExit = $null
  $killOutput = $null
  if ($timedOut) {
    [void]$stages.Add([ordered]@{name='TimedOut'; utc=([DateTime]::UtcNow).ToString('o')})
    $killOutput = (& "$env:SystemRoot\System32\taskkill.exe" /PID $pidValue /T /F 2>&1 | Out-String).Trim()
    $killExit = $LASTEXITCODE
    $exited = $process.WaitForExit(5000)
    [void]$stages.Add([ordered]@{name='TerminationAttempted'; utc=([DateTime]::UtcNow).ToString('o')})
  }
  if (-not $exited) { throw "Child remained active after timeout cleanup: $stageName pid=$pidValue" }
  $process.WaitForExit()
  $stdout = $outTask.GetAwaiter().GetResult()
  $stderr = $errTask.GetAwaiter().GetResult()
  $stageEnd = [DateTime]::UtcNow
  [void]$stages.Add([ordered]@{name='Completed'; utc=$stageEnd.ToString('o')})
  $parentStillActive = $null -ne (Get-Process -Id $pidValue -ErrorAction SilentlyContinue)
  $exitCode = $process.ExitCode
  $outcome = if ($timedOut) {'TimedOut'} elseif ($exitCode -eq 0) {'Success'} else {'FailedExit'}
  $record = [ordered]@{
    stage=$stageName; fileName=$psi.FileName; processArguments=@($processArguments)
    expectedArgumentCount=$expectedArgumentCount; actualPrelaunchArgumentCount=$processArguments.Length
    quotedArguments=$constructedArguments; pid=$pidValue; timeoutMs=$timeoutMs
    timedOut=$timedOut; killExitCode=$killExit; killOutput=$killOutput
    parentStillActive=$parentStillActive; exitCode=$exitCode; outcome=$outcome
    stdout=$stdout; stderr=$stderr
    startUtc=$stageStart.ToString('o'); endUtc=$stageEnd.ToString('o')
    elapsedMilliseconds=[math]::Round(($stageEnd-$stageStart).TotalMilliseconds)
    stages=@($stages)
  }
  $process.Dispose()
  return $record
}
