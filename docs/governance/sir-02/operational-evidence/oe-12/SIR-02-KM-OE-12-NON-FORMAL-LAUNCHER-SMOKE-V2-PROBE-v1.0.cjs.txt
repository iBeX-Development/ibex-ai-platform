const { spawn } = require('node:child_process');
const mode = process.argv[2];
if (mode === 'echo') {
  process.stdout.write('SMOKE_V2_STDOUT_MARKER\n');
  process.stdout.write('ARGV_JSON=' + JSON.stringify(process.argv) + '\n');
  process.stderr.write('SMOKE_V2_STDERR_MARKER\n');
} else if (mode === 'nonzero') {
  process.stdout.write('NONZERO_STDOUT_MARKER\n');
  process.stderr.write('NONZERO_STDERR_MARKER\n');
  process.exitCode = 23;
} else if (mode === 'tree') {
  const child = spawn(process.execPath, [__filename, 'hold-child'], { stdio: 'ignore', windowsHide: true });
  child.on('error', (error) => {
    process.stderr.write('TREE_CHILD_ERROR=' + error.message + '\n');
    process.exitCode = 24;
  });
  process.stdout.write('TREE_CHILD_PID=' + child.pid + '\n');
  process.stderr.write('TREE_STARTED\n');
  setInterval(() => {}, 1000);
} else if (mode === 'hold-child') {
  setInterval(() => {}, 1000);
} else {
  process.stderr.write('UNKNOWN_MODE\n');
  process.exitCode = 25;
}
