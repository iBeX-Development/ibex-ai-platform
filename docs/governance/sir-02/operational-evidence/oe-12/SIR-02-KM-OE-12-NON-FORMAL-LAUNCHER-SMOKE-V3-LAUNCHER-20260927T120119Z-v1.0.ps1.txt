function Set-Case([string]$id,[string]$expected,[string]$actual,[string]$result) {
  $caseResults[$id] = [ordered]@{id=$id; expected=$expected; actual=$actual; result=$result}
}

function Compare-StringArrays([string[]]$expected,[string[]]$actual) {
  if ($null -eq $expected -or $null -eq $actual -or $expected.Length -ne $actual.Length) { return $false }
  for ($i=0; $i -lt $expected.Length; $i++) {
    if (-not [string]::Equals($expected[$i],$actual[$i],[System.StringComparison]::Ordinal)) { return $false }
  }
  return $true
}

function Quote-WindowsArgument([AllowEmptyString()][string]$value) {
  if ($value.IndexOf([char]0) -ge 0) { throw 'NUL in process argument' }
  $builder = New-Object System.Text.StringBuilder
  $slash = [string][char]92
  $quote = [string][char]34
  [void]$builder.Append($quote)
  $backslashes = 0
  foreach ($ch in $value.ToCharArray()) {
    if ($ch -eq [char]92) { $backslashes++; continue }
    if ($ch -eq [char]34) {
      [void]$builder.Append(($slash * (2 * $backslashes + 1)))
      [void]$builder.Append($quote)
      $backslashes = 0
      continue
    }
    if ($backslashes -gt 0) {
      [void]$builder.Append(($slash * $backslashes))
      $backslashes = 0
    }
    [void]$builder.Append($ch)
  }
  if ($backslashes -gt 0) { [void]$builder.Append(($slash * (2 * $backslashes))) }
  [void]$builder.Append($quote)
  return $builder.ToString()
}

function Invoke-BoundedNode {
  param(
    [Parameter(Mandatory=$true)][string]$serializedArgumentVector,
    [Parameter(Mandatory=$true)][int]$expectedArgumentCount,
    [Parameter(Mandatory=$true)][int]$timeoutMs,
    [Parameter(Mandatory=$true)][string]$stageName
  )
  $expected = $script:scenarioVectors[$stageName]
  $t0 = [ordered]@{
    stage=$stageName
    utc=([DateTime]::UtcNow).ToString('o')
    serializedJson=$serializedArgumentVector
    expectedCount=$expectedArgumentCount
    decodedCount=$null
    decodedType=$null
    expectedVector=@($expected)
    decodedVector=@()
    perIndex=@()
    emptyPreserved=$null
    duplicatesPreserved=$null
    probePathPreserved=$false
    result='FAIL'
    failure=$null
  }
  [void]$t0History.Add($t0)
  try {
    if ([string]::IsNullOrWhiteSpace($serializedArgumentVector)) { throw 'T0 serialized JSON missing' }
    if ($null -eq $expected -or $expected.Length -ne $expectedArgumentCount) { throw 'T0 expected-vector count mismatch' }
    $parser = New-Object System.Web.Script.Serialization.JavaScriptSerializer
    $decodedRaw = $parser.DeserializeObject($serializedArgumentVector)
    if (-not ($decodedRaw -is [System.Array])) { throw 'T0 decoded root is not an array' }
    $t0.decodedType = $decodedRaw.GetType().FullName
    $t0.decodedCount = $decodedRaw.Length
    if ($decodedRaw.Length -ne $expectedArgumentCount) { throw 'T0 decoded count mismatch' }
    $processArguments = [System.Array]::CreateInstance([string], [int]$decodedRaw.Length)
    $checks = New-Object System.Collections.ArrayList
    for ($i=0; $i -lt $decodedRaw.Length; $i++) {
      if ($null -eq $decodedRaw[$i] -or -not ($decodedRaw[$i] -is [string])) { throw "T0 null/non-string at index $i" }
      $processArguments[$i] = [string]$decodedRaw[$i]
      $match = [string]::Equals($processArguments[$i],$expected[$i],[System.StringComparison]::Ordinal)
      [void]$checks.Add([ordered]@{index=$i; expected=$expected[$i]; decoded=$processArguments[$i]; match=$match})
      if (-not $match) { throw "T0 index mismatch at $i" }
    }
    $t0.perIndex = @($checks)
    $t0.decodedVector = @($processArguments)
    $t0.probePathPreserved = $processArguments.Length -gt 0 -and [string]::Equals($processArguments[0],$probe,[System.StringComparison]::Ordinal) -and (Test-Path -LiteralPath $probe -PathType Leaf)
    if (-not $t0.probePathPreserved) { throw 'T0 probe path missing or changed' }
    if ($stageName -eq 'echo') {
      $t0.emptyPreserved = $processArguments.Length -gt 4 -and $processArguments[4].Length -eq 0 -and $expected[4].Length -eq 0
      $t0.duplicatesPreserved = $processArguments.Length -gt 6 -and $processArguments[5] -ceq 'repeat-value' -and $processArguments[6] -ceq 'repeat-value'
      if (-not $t0.emptyPreserved -or -not $t0.duplicatesPreserved) { throw 'T0 empty or duplicate argument lost' }
    } else {
      $t0.emptyPreserved = 'Not Applicable'
      $t0.duplicatesPreserved = 'Not Applicable'
    }
    $t0.result = 'PASS'
  } catch {
    $t0.failure = $_.Exception.Message
    throw
  }
  if (-not (Test-Path -LiteralPath $node -PathType Leaf)) { throw "Pre-launch executable missing: $stageName" }
  if (-not (Test-Path -LiteralPath $probe -PathType Leaf)) { throw "Pre-launch probe missing: $stageName" }
  if ($processArguments.Length -eq 0) { throw "Pre-launch arguments empty: $stageName" }
  $quoted = New-Object System.Collections.Generic.List[string]
  for ($i=0; $i -lt $processArguments.Length; $i++) {
    $quoted.Add((Quote-WindowsArgument -value $processArguments[$i]))
  }
  $constructedArguments = [string]::Join(' ', $quoted.ToArray())
  if ([string]::IsNullOrWhiteSpace($constructedArguments)) { throw "Pre-launch Arguments empty: $stageName" }
  if (-not $constructedArguments.StartsWith((Quote-WindowsArgument -value $probe),[System.StringComparison]::Ordinal)) { throw "Pre-launch probe absent: $stageName" }
  $psi = New-Object System.Diagnostics.ProcessStartInfo
  $psi.FileName = $node
  $psi.Arguments = $constructedArguments
  $psi.WorkingDirectory = $repo
  $psi.UseShellExecute = $false
  $psi.CreateNoWindow = $true
  $psi.RedirectStandardOutput = $true
  $psi.RedirectStandardError = $true
  $psi.StandardOutputEncoding = [System.Text.UTF8Encoding]::new($false)
  $psi.StandardErrorEncoding = [System.Text.UTF8Encoding]::new($false)
  $process = New-Object System.Diagnostics.Process
  $process.StartInfo = $psi
  $stages = New-Object System.Collections.ArrayList
  $stageStart = [DateTime]::UtcNow
  [void]$stages.Add([ordered]@{name='Prepared'; utc=$stageStart.ToString('o')})
  if (-not $process.Start()) { throw "Process.Start returned false: $stageName" }
  [void]$stages.Add([ordered]@{name='Started'; utc=([DateTime]::UtcNow).ToString('o')})
  $pidValue = $process.Id
  $outTask = $process.StandardOutput.ReadToEndAsync()
  $errTask = $process.StandardError.ReadToEndAsync()
  $exited = $process.WaitForExit($timeoutMs)
  $timedOut = -not $exited
  $killExit = $null
  $killOutput = $null
  if ($timedOut) {
    [void]$stages.Add([ordered]@{name='TimedOut'; utc=([DateTime]::UtcNow).ToString('o')})
    $killOutput = (& "$env:SystemRoot\System32\taskkill.exe" /PID $pidValue /T /F 2>&1 | Out-String).Trim()
    $killExit = $LASTEXITCODE
    $exited = $process.WaitForExit(5000)
    [void]$stages.Add([ordered]@{name='TerminationAttempted'; utc=([DateTime]::UtcNow).ToString('o')})
  }
  if (-not $exited) { throw "Child remained active after timeout cleanup: $stageName pid=$pidValue" }
  $process.WaitForExit()
  $stdout = $outTask.GetAwaiter().GetResult()
  $stderr = $errTask.GetAwaiter().GetResult()
  $stageEnd = [DateTime]::UtcNow
  [void]$stages.Add([ordered]@{name='Completed'; utc=$stageEnd.ToString('o')})
  $parentStillActive = $null -ne (Get-Process -Id $pidValue -ErrorAction SilentlyContinue)
  $exitCode = $process.ExitCode
  $outcome = if ($timedOut) {'TimedOut'} elseif ($exitCode -eq 0) {'Success'} else {'FailedExit'}
  $record = [ordered]@{
    stage=$stageName
    fileName=$psi.FileName
    serializedTransport=$serializedArgumentVector
    decodedArgumentVector=@($processArguments)
    quotedArgumentRepresentations=@($quoted.ToArray())
    processStartInfoArguments=$psi.Arguments
    expectedArgumentCount=$expectedArgumentCount
    actualPrelaunchArgumentCount=$processArguments.Length
    pid=$pidValue
    timeoutMs=$timeoutMs
    timedOut=$timedOut
    killExitCode=$killExit
    killOutput=$killOutput
    parentStillActive=$parentStillActive
    exitCode=$exitCode
    outcome=$outcome
    stdout=$stdout
    stderr=$stderr
    startUtc=$stageStart.ToString('o')
    endUtc=$stageEnd.ToString('o')
    elapsedMilliseconds=[math]::Round(($stageEnd-$stageStart).TotalMilliseconds)
    stages=@($stages)
  }
  $process.Dispose()
  return $record
}
